{{/* Graphe des termes du glossaire (vue « à la Obsidian »). Nœuds = fiches de content/glossaire/termes/ dans la langue courante. Arêtes = liens /glossaire/termes// présents dans le corps des fiches (non orientés). Les étiquettes reprennent le title sans article initial. */}} {{- $nodes := slice -}} {{- $terms := slice -}} {{- range .Site.RegularPages -}} {{- if strings.Contains .RelPermalink "/glossaire/termes/" -}} {{- $terms = $terms | append . -}} {{- end -}} {{- end -}} {{- $ids := slice -}} {{/* Identifiant = dernier segment de l'URL (les fiches EN ont leurs propres slugs) */}} {{- range $terms }}{{ $ids = $ids | append (path.Base .RelPermalink) }}{{ end -}} {{- range $terms -}} {{- $id := path.Base .RelPermalink -}} {{- $links := slice -}} {{- range findRE `/glossaire/termes/[a-z0-9-]+` .RawContent -}} {{- $target := replaceRE `^.*/` "" . -}} {{- if and (ne $target $id) (in $ids $target) -}} {{- $links = $links | append $target -}} {{- end -}} {{- end -}} {{- $label := .Title | replaceRE `^(L['’]|Le |La |Les |The )` "" | strings.FirstUpper -}} {{- $nodes = $nodes | append (dict "id" $id "label" $label "url" .RelPermalink "links" ($links | uniq)) -}} {{- end -}}
